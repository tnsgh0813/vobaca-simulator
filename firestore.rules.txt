rules_version = '2';
// 바보카 BREAK 시뮬레이터 — Firestore 보안 규칙
// Firebase 콘솔 → Firestore Database → 규칙 탭에 이 내용 전체를 붙여넣고 「게시」
service cloud.firestore {
  match /databases/{database}/documents {
    // 각자 자기 기록(users/내 UID)만 읽고 쓸 수 있음. 게스트(익명 로그인)도 자기 문서만.
    match /users/{uid} {
      allow read, delete: if request.auth != null && request.auth.uid == uid;
      allow create, update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['d', 'name', 'at'])
        && request.resource.data.d is string && request.resource.data.d.size() < 300000
        && (!('name' in request.resource.data) || (request.resource.data.name is string && request.resource.data.name.size() <= 20))
        && (!('at' in request.resource.data) || request.resource.data.at is number);
    }
    // 관리자 목록: Firebase 콘솔에서 admins/<관리자 UID> 문서를 직접 만든다 (사이트에서는 만들 수 없음)
    function isAdmin() { return request.auth != null && exists(/databases/$(database)/documents/admins/$(request.auth.uid)); }
    match /admins/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if false;
    }
    // 효과음 · 배경음 교체: 누구나 읽고(모든 방문자에게 적용), 관리자만 바꿈
    match /sound/{id} {
      allow read: if true;
      allow delete: if isAdmin();
      allow create, update: if isAdmin() && id.size() <= 40
        && request.resource.data.keys().hasOnly(['name', 'mime', 'size', 'chunks', 'ver', 'vol', 'at', 'by'])
        && request.resource.data.chunks is int && request.resource.data.chunks <= 10
        && (!('vol' in request.resource.data) || (request.resource.data.vol is number && request.resource.data.vol >= 0 && request.resource.data.vol <= 3));
    }
    match /soundChunks/{id} {
      allow read: if true;
      allow delete: if isAdmin();
      allow create, update: if isAdmin() && request.resource.data.keys().hasOnly(['b'])
        && request.resource.data.b is bytes && request.resource.data.b.size() <= 1000000;
    }
    // 온라인 멀티 대전: 방(rooms/코드)과 선택 기록(rooms/코드/moves/번호)
    function roomKeys() { return ['code', 'host', 'hostName', 'guest', 'guestName', 'ver', 'status', 'hostDeck', 'guestDeck', 'hostReady', 'guestReady', 'seed', 'conceded', 'at', 'hb0', 'hb1', 'startedAt']; }
    function roomOk(d) {
      return d.keys().hasOnly(roomKeys())
        && d.hostName is string && d.hostName.size() <= 20 && d.guestName is string && d.guestName.size() <= 20
        && d.hostDeck is string && d.hostDeck.size() < 4000 && d.guestDeck is string && d.guestDeck.size() < 4000
        && d.status in ['open', 'playing', 'ended', 'closed'];
    }
    match /rooms/{code} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && code.matches('^[A-Z0-9]{6}$') && roomOk(request.resource.data)
        && request.resource.data.code == code && request.resource.data.host == request.auth.uid && request.resource.data.guest == null;
      allow update: if request.auth != null && roomOk(request.resource.data)
        && request.resource.data.host == resource.data.host && request.resource.data.code == resource.data.code
        && (
          // 방장 · 참가자는 방 정보를 바꿀 수 있음 (참가자는 다른 사람을 참가자로 넣을 수 없음)
          resource.data.host == request.auth.uid
          || (resource.data.guest == request.auth.uid && (request.resource.data.guest == request.auth.uid || request.resource.data.guest == null))
          // 빈 자리에 내가 들어가기
          || (resource.data.guest == null && resource.data.status == 'open' && request.resource.data.guest == request.auth.uid)
        );
      allow delete: if request.auth != null && resource.data.host == request.auth.uid;
      function isMember() {
        let r = get(/databases/$(database)/documents/rooms/$(code)).data;
        return request.auth != null && (r.host == request.auth.uid || r.guest == request.auth.uid);
      }
      match /moves/{n} {
        allow read: if isMember();
        allow create: if isMember() && n.matches('^[0-9]{6}$')
          && request.resource.data.keys().hasOnly(['s', 'p', 't', 'v', 'h', 'by'])
          && request.resource.data.by == request.auth.uid && request.resource.data.s is int
          && request.resource.data.v is string && request.resource.data.v.size() < 2000;
        allow update, delete: if false;
      }
    }
    // 그 밖의 모든 곳은 막음
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
